# Revocation is optional and independent of PDF handling and the GUI.
if(NOT JPDF_DESK_WITH_OPENSSL)
    return()
endif()
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(jpdf-desk-revocation STATIC src/revocation.cpp)
add_library(JPDFDesk::Revocation ALIAS jpdf-desk-revocation)
target_include_directories(jpdf-desk-revocation PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(jpdf-desk-revocation PUBLIC jpdf-desk-revocation-types jpdf-desk-crypto-support PRIVATE OpenSSL::Crypto)

if(TARGET Qt6::Network)
    add_library(jpdf-desk-revocation-http STATIC src/http_transport.cpp)
    add_library(JPDFDesk::RevocationHttp ALIAS jpdf-desk-revocation-http)
    target_link_libraries(jpdf-desk-revocation-http PUBLIC jpdf-desk-revocation Qt6::Network)
    if(TARGET Qt6::QTlsBackendOpenSSLPlugin)
        target_link_libraries(jpdf-desk-revocation-http PRIVATE Qt6::QTlsBackendOpenSSLPlugin)
        target_compile_definitions(jpdf-desk-revocation-http PRIVATE JPDF_REVOCATION_STATIC_TLS)
    endif()
endif()
